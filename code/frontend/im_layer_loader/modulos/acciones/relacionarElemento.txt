def mi_funcion(idFeatureForm, layerNameMapa, nombreAtributoMapa, layerNameFormulario, nombreAtributoFormulario, selected_features):

    try:
        capa.selectionChanged.disconnect(conexion)
    except Exception as error:
        print(f'No pude desconectar {type(error).__name__} : {error}')

    cant_sel = len(selected_features)
    featureId=0
    if cant_sel > 0:
        featureId = selected_features[cant_sel-1]
        layer = QgsProject.instance().mapLayersByName(layerNameMapa)[0]
        feature = layer.getFeature(featureId)
        if feature.isValid():
            dato = feature[nombreAtributoMapa]
            layerForm = QgsProject.instance().mapLayersByName(layerNameFormulario)[0]
            if idFeatureForm > 0:
                featureForm = layerForm.getFeature(idFeatureForm)
                featureForm.setAttribute(featureForm.fieldNameIndex(nombreAtributoFormulario), dato)
                layerForm.updateFeature(featureForm)
                qgs = QgsProject.instance()
                qgs.layerTreeRoot().findLayer(layerForm.id()).setItemVisibilityChecked(True)
                iface.layerTreeView().setCurrentLayer(layerForm)
                iface.openFeatureForm(layerForm, featureForm)
            else:
                if layerForm.isEditable():
                    x_var_global_featrue.setAttribute(x_var_global_featrue.fieldNameIndex(nombreAtributoFormulario), dato)
                    layerForm.startEditing()
                    if (esta_en_capa(layerForm, x_var_global_featrue)):
                        #quito el punto anterior para que no quede doble 
                        layerForm.deleteFeature(x_var_global_featrue.id())

                    #Agrego el punto a la capa para el openFeatureForm    
                    layerForm.addFeature(x_var_global_featrue)
                    #me quedo esperando que lo cierre el flujo o lo cancelen
                    valores_compartidos["cancelar_flujo"] = True
                    if (iface.openFeatureForm ( layerForm, x_var_global_featrue, False)):
                        layerForm.triggerRepaint()
                        iface.layerTreeView().setCurrentLayer(layerForm)
                        #mato la variable
                        globals().pop('x_var_global_featrue')

                    else:                        
                        if (valores_compartidos["cancelar_flujo"]):
                        
                            if (esta_en_capa(layerForm, x_var_global_featrue)):
                        
                                layerForm.deleteFeature(x_var_global_featrue.id())
                                iface.layerTreeView().setCurrentLayer(layerForm)

                        layerForm.triggerRepaint()
                        

def esta_en_capa(capa, punto):
    
    r = QgsFeatureRequest(punto.id())
    feature = capa.getFeatures(r)

    if next(feature, None) is not None:
        esta = True
    else:
        esta = False

    return esta

def obtener_dialogo_activo():

    app = QApplication.instance()
    for widget in app.topLevelWidgets():
        if isinstance(widget, QDialog) and widget.isVisible() and widget.isActiveWindow() and widget.windowTitle() == layerNameFormulario + " - Atributos del objeto espacial":
            return widget
    return None


def son_mismo_dia(fecha1, fecha2):

   fecha1_sin_hora = QDate(fecha1.date())
   fecha2_sin_hora = QDate(fecha2.date())
   return fecha1_sin_hora == fecha2_sin_hora


def obtener_valores_campos(dialogo):

    for campo in capa_luminaria.fields():
        nombreAtributo = campo.name()

        actualizar_atr_punto = True
        evaluar_dato = dialogo.findChild(QLineEdit, nombreAtributo)
        if (evaluar_dato is not None):
            dato = evaluar_dato.text()
            if dato == 'NULL' or dato == '':
                dato = None
        else:
            evaluar_dato = dialogo.findChild(QSpinBox, nombreAtributo)
            if (evaluar_dato is not None):
                if evaluar_dato.value() != evaluar_dato.minimum():
                    dato = evaluar_dato.value()
                else:
                    dato = None
            else:
                evaluar_dato = dialogo.findChild(QDateTimeEdit, nombreAtributo)         
                if (evaluar_dato is not None):
                    if evaluar_dato.dateTime().time().hour() == 0:
                        dato = evaluar_dato.dateTime().toString("yyyy-MM-dd") + "T03:00:00.000Z"
                    else:
                        fecha = evaluar_dato.dateTime().toString("yyyy-MM-dd HH:mm:ss")
                        if son_mismo_dia(QDateTime.fromString(fecha, "yyyy-MM-dd HH:mm:ss"), QDateTime.currentDateTime()):
                            dato = None
                        else:
                            dato = evaluar_dato.dateTime().toString("yyyy-MM-dd") + "T03:00:00.000Z"
                else:            
                    evaluar_dato = dialogo.findChild(QComboBox, nombreAtributo)             
                    if (evaluar_dato is not None): 
                        if evaluar_dato.currentText() == '(NULL)':
                            dato = None
                        else:
                            dato = evaluar_dato.currentData()
                    else:
                        actualizar_atr_punto = False    
        if actualizar_atr_punto:
            x_var_global_featrue.setAttribute(x_var_global_featrue.fieldNameIndex(nombreAtributo), dato)
